cmake_minimum_required(VERSION 3.22)
project(LinVitaShaderRegression LANGUAGES C CXX OBJCXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
get_filename_component(VITA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_OSX_DEPLOYMENT_TARGET 13.3)
set(FMT_TEST OFF)
add_subdirectory(${VITA_ROOT}/external/fmt fmt EXCLUDE_FROM_ALL)
set(SPDLOG_FMT_EXTERNAL ON)
add_subdirectory(${VITA_ROOT}/external/spdlog spdlog EXCLUDE_FROM_ALL)
set(ENABLE_GLSLANG_BINARIES OFF)
set(ENABLE_HLSL OFF)
set(GLSLANG_TESTS OFF)
set(ENABLE_OPT OFF)
add_subdirectory(${VITA_ROOT}/external/glslang glslang EXCLUDE_FROM_ALL)
set(SPIRV_CROSS_CLI OFF)
set(SPIRV_CROSS_ENABLE_TESTS OFF)
set(SPIRV_CROSS_ENABLE_HLSL OFF)
set(SPIRV_CROSS_ENABLE_MSL ON)
set(SPIRV_CROSS_ENABLE_CPP OFF)
set(SPIRV_CROSS_ENABLE_REFLECT OFF)
set(SPIRV_CROSS_ENABLE_C_API OFF)
set(SPIRV_CROSS_ENABLE_UTIL OFF)
set(SPIRV_CROSS_SKIP_INSTALL ON)
add_subdirectory(${VITA_ROOT}/external/SPIRV-Cross cross EXCLUDE_FROM_ALL)
# Only shader-relevant utilities are needed; no emulator runtime is substituted.
add_library(util STATIC ${VITA_ROOT}/vita3k/util/src/float_to_half.cpp ${VITA_ROOT}/vita3k/util/src/string_utils.cpp)
target_include_directories(util PUBLIC ${VITA_ROOT}/vita3k/util/include ${VITA_ROOT}/vita3k/mem/include ${VITA_ROOT}/external/boost ${VITA_ROOT}/external/half/include ${VITA_ROOT}/external/xxHash ${VITA_ROOT}/external/sdl/include)
target_link_libraries(util PUBLIC fmt spdlog)
add_subdirectory(${VITA_ROOT}/vita3k/features features)
add_subdirectory(${VITA_ROOT}/vita3k/gxm gxm)
add_library(googletest STATIC ${VITA_ROOT}/external/googletest/googletest/src/gtest_main.cc ${VITA_ROOT}/external/googletest/googletest/src/gtest-all.cc)
target_include_directories(googletest PUBLIC ${VITA_ROOT}/external/googletest/googletest/include PRIVATE ${VITA_ROOT}/external/googletest/googletest)
add_subdirectory(${VITA_ROOT}/vita3k/shader shader)
if(SHADER_RECOMPILER_OVERRIDE)
    get_target_property(shader_sources shader SOURCES)
    list(REMOVE_ITEM shader_sources src/spirv_recompiler.cpp)
    list(APPEND shader_sources "${SHADER_RECOMPILER_OVERRIDE}")
    set_property(TARGET shader PROPERTY SOURCES "${shader_sources}")
endif()
target_link_options(shader-tests PRIVATE -Wl,-dead_strip)
add_executable(shader-replay replay.mm)
target_link_libraries(shader-replay PRIVATE shader spirv-cross-msl "-framework Foundation" "-framework Metal")
target_link_options(shader-replay PRIVATE -Wl,-dead_strip)
